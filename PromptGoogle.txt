// Helper to write to Doorbell
fn xhci_ring_doorbell(u64 db_base, i64 slot_id, i64 ep_idx) {
    // Doorbell Register Array starts at db_base.
    // DB Register for Slot N is at: db_base + (4 * slot_id)
    ptr<u32> db_reg = ptr<u32>(db_base + (slot_id * 4));
    
    print("DB Ring: Slot=", i64(slot_id), " EP=", i64(ep_idx), " Addr=", u64(db_reg), "\n");
    
    // Flush cache to ensure TRBs are in RAM before HC sees doorbell
    rt_wbinvd();
    
    // Write Endpoint Context Index (Target) to Doorbell
    volatile_store(db_reg, u32(ep_idx));
}

// Queue a No-Op TRB to verify the ring is working
fn xhci_queue_noop(ptr<u64> ring_base, ptr<i64> idx_ptr, ptr<i64> cycle_ptr, u64 db_base, i64 slot_id, i64 ep_idx) {
    i64 idx = *idx_ptr;
    i64 cycle = *cycle_ptr;
    
    print("Q_NOOP: Idx=", idx, " Cyc=", cycle, "\n");
    
    // No-Op TRB: Type 23 (0x17), IOC=1
    // Data (param) = 0
    // Status = 0
    // Control = (Type=23<<10) | (IOC=1<<5) | (Cycle<<0)
    
    u32 d3 = u32((23 << 10) | (1 << 5) | cycle);
    
    volatile_store(ring_base + (idx * 16) + 0, u64(0)); // d0, d1
    volatile_store(ring_base + (idx * 16) + 8, u64(0)); // d2 (u32), d3 (u32) low clear
    // Write d3 last to set cycle bit atomically-ish
    ptr<u32> d3_ptr = ptr<u32>(u64(ring_base) + (idx * 16) + 12);
    volatile_store(d3_ptr, d3);
    
    idx = idx + 1;
    if (idx >= 256) {
        // Link TRB handling would go here, assuming linear ring for test
        idx = 0;
        cycle = cycle ^ 1;
    }
    
    *idx_ptr = idx;
    *cycle_ptr = cycle;
    
    xhci_ring_doorbell(db_base, slot_id, ep_idx);
}
